> **今日目标**：看穿任何「奇怪等号」，能在 Code Review 中一眼识别隐式转换陷阱，掌握安全类型转换的工程模式。

## 一、`==` vs `===` 完整对比

### 1.1 严格相等（`===`）— 无类型转换

```javascript
// 严格相等：类型不同直接返回 false
console.log(1 === '1');      // false（类型不同）
console.log(null === undefined); // false（类型不同）
console.log(NaN === NaN);    // false（NaN 特殊规则）

// 推荐：绝大多数情况下使用 ===
```

### 1.2 抽象相等（`==`）— 有类型转换，规则复杂

```javascript
// 同类型：与 === 相同
console.log(1 == 1);         // true
console.log('a' == 'a');     // true

// null/undefined 互等，且只等于彼此
console.log(null == undefined);  // true  ← 有用！
console.log(null == 0);          // false
console.log(null == false);      // false
console.log(undefined == 0);    // false
console.log(undefined == false); // false

// Number vs String：String 转 Number
console.log(1 == '1');       // true  (ToNumber('1') = 1)
console.log(0 == '');        // true  (ToNumber('') = 0)
console.log(0 == '0');       // true  (ToNumber('0') = 0)

// Boolean vs 其他：Boolean 先转 Number
console.log(true == 1);      // true  (ToNumber(true) = 1)
console.log(false == 0);     // true  (ToNumber(false) = 0)
console.log(true == '1');    // true  (1 == '1' → 1 == 1)
console.log(false == '');    // true  (0 == '' → 0 == 0)

// Object vs 原始值：对象先 ToPrimitive
console.log([] == false);    // true  （经历 3 次转换！）
console.log([] == 0);        // true
console.log('' == false);    // true
console.log([1] == 1);       // true
console.log([1,2] == '1,2'); // true
```

> [!WARNING]
> **团队规范**：除了 `x == null`（同时检测 null 和 undefined）这一惯用法之外，其他场景一律禁用 `==`，强制使用 `===`。这是所有主流 ESLint 规则集的共识。

### 1.3 唯一推荐的 `==` 使用场景

```javascript
// ✅ 唯一合法的 == 用法：同时检测 null 和 undefined
function processValue(val) {
  if (val == null) {
    // val 是 null 或 undefined，均走此分支
    return defaultValue;
  }
  return val;
}

// 等价的严格写法（更啰嗦）
if (val === null || val === undefined) { }

// 现代写法（更优雅）
const result = val ?? defaultValue; // 空值合并运算符
```

## 二、`+` 运算符的二义性陷阱

`+` 是 JavaScript 中最危险的运算符，兼具「数字加法」和「字符串拼接」两种语义：

```javascript
// 规则：有任一操作数是 String → 字符串拼接；否则数字加法

// 数字 + 数字 → 加法
console.log(1 + 2);          // 3

// 字符串 + 任何值 → 拼接
console.log('1' + 2);        // '12'
console.log(1 + '2');        // '12'
console.log('' + 1);         // '1'（快速转字符串，但不推荐）

// 对象参与 + → 先 ToPrimitive(hint: default)
console.log({} + []);        // '[object Object]'（{} 的 toString）
console.log([] + {});        // '[object Object]'（同上，[]→'' 先）
console.log([] + []);        // ''（都转为空字符串）

// ⚠️ 经典迷惑题：{} + [] 在不同上下文的不同结果
// 在语句位置：{} 被解析为空代码块，+[] 是一元 + 操作
{} + []  // 0（命令行中执行）
// 在表达式位置：{} 是对象字面量
({}) + [] // '[object Object]'

// ✅ 工程建议：数字运算永远显式转换
const a = '5', b = '3';
console.log(a + b);           // '53'（错误！）
console.log(Number(a) + Number(b)); // 8（正确）
console.log(+a + +b);         // 8（简洁，但可读性差）
```

## 三、对象参与运算的转换规则

```javascript
// ToPrimitive 转换路径（hint: default/number 时）：
// 1. 检查 [Symbol.toPrimitive] 方法
// 2. 若无，调用 valueOf()，若返回原始值则使用
// 3. 若无，调用 toString()，若返回原始值则使用

// 默认行为
const obj = {};
console.log(obj.valueOf());  // {} 本身（非原始值，继续）
console.log(obj.toString()); // '[object Object]'

// 数组的行为
const arr = [1, 2, 3];
console.log(arr.valueOf());  // [1,2,3] 本身（非原始值，继续）
console.log(arr.toString()); // '1,2,3'（join(',')）

// 自定义转换顺序
const money = {
  amount: 100,
  currency: 'CNY',

  // 优先级最高：覆盖所有 hint
  [Symbol.toPrimitive](hint) {
    console.log('hint:', hint);
    if (hint === 'number') return this.amount;
    if (hint === 'string') return `${this.amount} ${this.currency}`;
    return this.amount; // 'default' 同 number
  }
};

console.log(+money);          // hint: number → 100
console.log(`${money}`);      // hint: string → '100 CNY'
console.log(money + 50);      // hint: default → 150
console.log(money > 50);      // hint: number → true
```

### 3.1 Date 对象的特殊行为

```javascript
// Date 是唯一一个 hint: default 等同于 hint: string 的内置对象
const date = new Date('2024-01-01');

// hint: default（被当作 string 处理）
console.log(date + '');      // 'Mon Jan 01 2024 08:00:00 GMT+0800'

// hint: number
console.log(+date);          // 1704038400000（时间戳毫秒数）

// 正确获取时间戳
console.log(Date.now());         // 推荐
console.log(new Date().getTime()); // 也可以
console.log(+new Date());        // 可用，但不直观
```

## 四、ToBoolean 转换：6 大假值全解

```javascript
// 仅以下 6 个值（加 document.all 这个特殊历史遗留）为假：
const falsyValues = [false, 0, -0, 0n, '', null, undefined, NaN];

falsyValues.forEach(v => {
  console.log(`Boolean(${JSON.stringify(v) ?? String(v)}) = ${Boolean(v)}`);
});

// 真值陷阱（初学者常踩）
console.log(Boolean([]));     // true  ← 空数组是真值！
console.log(Boolean({}));     // true  ← 空对象是真值！
console.log(Boolean('0'));    // true  ← 字符串 '0' 是真值！
console.log(Boolean('false')); // true ← 字符串 'false' 是真值！
console.log(Boolean(-1));     // true  ← 负数（非0）是真值！
console.log(Boolean(Infinity)); // true

// ⚠️ 常见 bug：用对象/数组作布尔判断
const settings = {};
if (settings) {  // 始终为 true，即使是空对象！
  console.log('settings exist'); // 总是执行
}

// ✅ 正确判断：检查具体属性
if (settings && Object.keys(settings).length > 0) {
  console.log('settings has properties');
}
```

## 五、最常见隐式转换陷阱 Top 10

```javascript
// 陷阱 1：+[] 和 +{} 的结果
console.log(+[]);   // 0（ToNumber(ToPrimitive([])) = ToNumber('') = 0）
console.log(+{});   // NaN（ToNumber('[object Object]') = NaN）

// 陷阱 2：模板字符串中的对象
const obj = { x: 1 };
console.log(`value: ${obj}`);  // 'value: [object Object]'（调用 toString）

// 陷阱 3：比较运算中的字符串
console.log('10' > '9');   // false！（字符串按字典序，'1' < '9'）
console.log(10 > 9);       // true（数字比较）

// 陷阱 4：数组的 join 在加法中
console.log([1] + [2]);    // '12'（[1].toString() = '1', [2].toString() = '2'）
console.log([1,2] + [3]);  // '1,23'

// 陷阱 5：switch 使用严格相等
const val = '1';
switch (val) {
  case 1:   // 严格相等：'1' !== 1，不匹配！
    console.log('number 1');
    break;
  case '1': // 匹配
    console.log('string 1'); // ← 执行这里
}

// 陷阱 6：parseInt 的隐藏参数
console.log(['1','2','3'].map(parseInt));
// 实际调用：parseInt('1',0), parseInt('2',1), parseInt('3',2)
// 结果：[1, NaN, NaN]（不是 [1,2,3]！）
// ✅ 正确：['1','2','3'].map(Number)

// 陷阱 7：if 语句中的对象
const response = { success: false };
if (response.success) {           // false，进入 else
  console.log('success');
} else {
  console.log('failed');          // 执行这里
}
// ✅ 没问题，但要小心对象本身 vs 对象属性的区别
if (response) {                   // true！对象本身是真值
  console.log('response exists'); // 总是执行
}

// 陷阱 8：undefined 参与数字运算
let count;
count += 1;  // undefined + 1 = NaN
console.log(count); // NaN

// 陷阱 9：空数组 == false
console.log([] == false);   // true（特殊的连环转换）
console.log([] == !false);  // true（!false = true，[] == true → 0 == 1 → false）
// 等等，其实 [] == !false 是 false！
console.log([] == !false);  // false！（!false = true，ToNumber([]) = 0，ToNumber(true) = 1，0 != 1）

// 陷阱 10：字符串索引访问
const str = 'hello';
console.log(str[0]);         // 'h'
console.log(str[-1]);        // undefined（不是最后一个字符！）
// ✅ 现代写法
console.log(str.at(-1));     // 'o'（ES2022+）
```

## 六、防御性编程：杜绝隐式转换 bug

### 6.1 TypeScript 的终极方案

```typescript
// 类型系统从根本上消灭隐式转换问题
function add(a: number, b: number): number {
  return a + b; // 编译期保证 a, b 都是 number
}
```

### 6.2 JavaScript 中的防御策略

```javascript
// 策略1：显式转换函数
const toNum = (v) => {
  const n = Number(v);
  if (Number.isNaN(n)) throw new TypeError(`Cannot convert ${v} to number`);
  return n;
};

// 策略2：运行时类型断言
function assertNumber(v, name = 'value') {
  if (typeof v !== 'number' || Number.isNaN(v)) {
    throw new TypeError(`${name} must be a valid number, got ${typeof v}: ${v}`);
  }
}

function safeAdd(a, b) {
  assertNumber(a, 'a');
  assertNumber(b, 'b');
  return a + b;
}

// 策略3：ESLint 规则强制
// .eslintrc.js 配置：
// "eqeqeq": ["error", "always", { "null": "ignore" }]
// "no-implicit-coercion": ["error", { "boolean": false, "string": false, "number": true }]
```
