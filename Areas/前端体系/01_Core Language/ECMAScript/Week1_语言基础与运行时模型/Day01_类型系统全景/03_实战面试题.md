## 一、知识点速查总结表

| 类型 | typeof 返回值 | 假值(falsy)? | 可作 key? | 可 JSON 序列化? | 特殊注意 |
|:---|:---|:---:|:---:|:---:|:---|
| `undefined` | `"undefined"` | ✅ | ✅(变为"undefined") | ❌(被忽略/null) | 未赋值变量的默认值 |
| `null` | `"object"` ⚠️ | ✅ | ✅(变为"null") | ✅(输出`null`) | typeof 历史 bug |
| `boolean` | `"boolean"` | ✅(false) | ✅ | ✅ | 6 个假值之一 |

## 二、工程陷阱案例

### 陷阱 1：typeof null 导致的空值判断 bug

**问题代码**：
```javascript
// 某同事写的「判断是否是对象」的工具函数
function isObject(val) {
  return typeof val === 'object';
}

// 意图：只有对象才进入这个分支
if (isObject(config)) {
  config.timeout = 3000; // 💥 当 config = null 时，报错！
}
```

**原因分析**：
```javascript
console.log(typeof null);    // 'object'
console.log(isObject(null)); // true ← 错误！null 通过了检测
// 然后 null.timeout = ... → TypeError: Cannot set properties of null
```

**正确写法**：
```javascript
// ✅ 方案1：加 null 排除
function isObject(val) {
  return val !== null && typeof val === 'object';
}

// ✅ 方案2：更严格（排除数组、Date 等）
function isPlainObject(val) {
  return val !== null &&
    typeof val === 'object' &&
    Object.getPrototypeOf(val) === Object.prototype;
}

// ✅ 方案3：可选链（最优雅）
config?.timeout ?? (config = { timeout: 3000 });
```

### 陷阱 2：Symbol 键不被 JSON.stringify 序列化

**问题代码**：
```javascript
const SESSION_ID = Symbol('sessionId');

const request = {
  url: '/api/data',
  [SESSION_ID]: 'abc123',  // 用 Symbol 键存储 session
  body: { query: 'test' }
};

// 发送请求前序列化
const payload = JSON.stringify(request);
console.log(payload);
// 输出：{"url":"/api/data","body":{"query":"test"}}
// ❌ sessionId 完全丢失！
```

**原因**：`JSON.stringify` 会跳过所有 Symbol 键，这是规范定义的行为（Clause 25.5.2.2）。

**正确做法**：
```javascript
// ✅ 如果需要序列化，使用 String 键
const request = {
  url: '/api/data',
  _sessionId: 'abc123',     // 内部约定，或用 Symbol 管理但序列化时手动提取
  body: { query: 'test' }
};

// ✅ 或者自定义 toJSON
const request2 = {
  url: '/api/data',
  [SESSION_ID]: 'abc123',
  toJSON() {
    return {
      url: this.url,
      sessionId: this[SESSION_ID], // 手动包含 Symbol 值
      body: this.body
    };
  }
};
```

### 陷阱 3：BigInt 与 Number 混合运算的 TypeError

**问题代码**：
```javascript
// 从数据库取出的 64 位用户 ID（超过安全整数范围）
const userId = BigInt('9007199254740993');
const PAGE_SIZE = 20; // 普通 Number

// 计算分页偏移
const offset = userId % PAGE_SIZE; // 💥 TypeError!
// TypeError: Cannot mix BigInt and other types, use explicit conversions
```

**正确写法**：
```javascript
// ✅ 方案1：统一使用 BigInt
const PAGE_SIZE = 20n;
const offset = userId % PAGE_SIZE; // 3n

// ✅ 方案2：转换 Number（仅在值安全范围内时）
const offsetNum = Number(userId % BigInt(PAGE_SIZE)); // 3

// ✅ 方案3：封装安全工具
function bigintMod(a, b) {
  return a % BigInt(b);
}
```

### 题目 1：实现精确的 `deepTypeOf` 函数

**要求**：返回值的精确类型字符串，覆盖所有内置类型，结果全部小写。

```javascript
// 题目要求覆盖：
// deepTypeOf(null)        → 'null'
// deepTypeOf(undefined)   → 'undefined'
// deepTypeOf(42)          → 'number'
// deepTypeOf(NaN)         → 'nan'（额外要求）
// deepTypeOf(Infinity)    → 'infinity'（额外要求）
// deepTypeOf(42n)         → 'bigint'
// deepTypeOf(Symbol())    → 'symbol'
// deepTypeOf([])          → 'array'
// deepTypeOf({})          → 'object'
// deepTypeOf(new Date())  → 'date'
// deepTypeOf(/re/)        → 'regexp'
// deepTypeOf(new Map())   → 'map'
// deepTypeOf(new Set())   → 'set'
// deepTypeOf(Promise.resolve()) → 'promise'
// deepTypeOf(function(){}) → 'function'
// deepTypeOf(class{})     → 'class'（额外挑战）
```

**完整答案**：
```javascript
function deepTypeOf(value) {
  // 处理 null（typeof 会返回 'object'，需提前处理）
  if (value === null) return 'null';

  // 处理原始类型（除 object/function 外）
  const t = typeof value;
  if (t === 'undefined') return 'undefined';
  if (t === 'boolean') return 'boolean';
  if (t === 'string') return 'string';
  if (t === 'symbol') return 'symbol';
  if (t === 'bigint') return 'bigint';

  // 处理 number 的特殊值
  if (t === 'number') {
    if (Number.isNaN(value)) return 'nan';
    if (!Number.isFinite(value)) return 'infinity';
    return 'number';
  }

  // 处理 function（区分普通函数和 class）
  if (t === 'function') {
    // class 的 toString 以 'class' 开头
    const str = Function.prototype.toString.call(value);
    if (str.startsWith('class')) return 'class';
    return 'function';
  }

  // 处理对象类型（使用 Object.prototype.toString 精确判断）
  return Object.prototype.toString.call(value).slice(8, -1).toLowerCase();
}

// 测试
console.log(deepTypeOf(null));           // 'null'      ✅
console.log(deepTypeOf(undefined));      // 'undefined' ✅
console.log(deepTypeOf(42));             // 'number'    ✅
console.log(deepTypeOf(NaN));            // 'nan'       ✅
console.log(deepTypeOf(Infinity));       // 'infinity'  ✅
console.log(deepTypeOf(42n));            // 'bigint'    ✅
console.log(deepTypeOf(Symbol()));       // 'symbol'    ✅
console.log(deepTypeOf([]));             // 'array'     ✅
console.log(deepTypeOf({}));             // 'object'    ✅
console.log(deepTypeOf(new Date()));     // 'date'      ✅
console.log(deepTypeOf(/re/));           // 'regexp'    ✅
console.log(deepTypeOf(new Map()));      // 'map'       ✅
console.log(deepTypeOf(Promise.resolve())); // 'promise' ✅
console.log(deepTypeOf(function(){}));   // 'function'  ✅
console.log(deepTypeOf(class MyClass{})); // 'class'    ✅
```

### 题目 2：解释 0.1 + 0.2 !== 0.3 的原因，并给出精度安全的比较方案

**答案**：

**原因**：JavaScript 的 `number` 类型使用 **IEEE 754 binary64（双精度浮点）** 格式。`0.1` 和 `0.2` 在二进制浮点中都是无限循环小数，存储时被截断，导致相加结果不等于精确的 `0.3`。

```
0.1 的二进制表示（无限循环）：
0.0001100110011001100110011001100110011001100110011... (无穷)

存储为 64 位时截断为：
0.1000000000000000055511151231257827021181583404541015625

0.1 + 0.2 的实际计算结果：
0.30000000000000004 (不等于 0.3)
```

**精度安全的比较方案**：

```javascript
// 方案1：使用 Number.EPSILON（适合一般浮点比较）
function floatEquals(a, b, epsilon = Number.EPSILON) {
  return Math.abs(a - b) < epsilon * Math.max(Math.abs(a), Math.abs(b), 1);
}
console.log(floatEquals(0.1 + 0.2, 0.3)); // true

// 方案2：转为整数运算（适合货币计算）
function addMoney(a, b) {
  // 将元转为分（整数），避免浮点误差
  return (Math.round(a * 100) + Math.round(b * 100)) / 100;
}
console.log(addMoney(0.1, 0.2)); // 0.3

// 方案3：使用 BigInt（精度要求最高）
function preciseMul(a, b, decimals = 2) {
  const factor = 10n ** BigInt(decimals);
  const aBig = BigInt(Math.round(a * Number(factor)));
  const bBig = BigInt(Math.round(b * Number(factor)));
  return Number(aBig + bBig) / Number(factor);
}
console.log(preciseMul(0.1, 0.2)); // 0.3

// 方案4：toFixed + parseFloat（简单场景）
console.log(parseFloat((0.1 + 0.2).toFixed(10))); // 0.3
```

### 题目 3：实现 `SafeNumber` 类，防止精度丢失

```javascript
class SafeNumber {
  #value; // 以分（整数）存储

  constructor(yuan) {
    if (typeof yuan !== 'number' && typeof yuan !== 'string') {
      throw new TypeError('SafeNumber: 参数必须是 number 或 string');
    }
    // 转为整数分存储（避免浮点误差）
    this.#value = Math.round(parseFloat(yuan) * 100);
  }

  add(other) {
    const result = new SafeNumber(0);
    result.#value = this.#value + other.#value;
    return result;
  }

  sub(other) {
    const result = new SafeNumber(0);
    result.#value = this.#value - other.#value;
    return result;
  }

  mul(factor) {
    const result = new SafeNumber(0);
    result.#value = Math.round(this.#value * factor);
    return result;
  }

  toNumber() {
    return this.#value / 100;
  }

  toString() {
    return (this.#value / 100).toFixed(2);
  }

  [Symbol.toPrimitive](hint) {
    if (hint === 'number') return this.toNumber();
    return this.toString();
  }
}

// 测试
const price = new SafeNumber('0.1');
const tax = new SafeNumber('0.2');
const total = price.add(tax);
console.log(total.toNumber()); // 0.3 ✅（精确）
console.log(`总价：${total}`); // '总价：0.30'
console.log(+total);           // 0.3 ✅（通过 Symbol.toPrimitive）
```

### 题目 4：Symbol 实现真正的私有状态

```javascript
// 实现一个 EventEmitter，用 Symbol 保护内部状态不被外部访问

const _listeners = Symbol('listeners');
const _maxListeners = Symbol('maxListeners');

class EventEmitter {
  constructor(maxListeners = 10) {
    this[_listeners] = new Map();
    this[_maxListeners] = maxListeners;
  }

  on(event, listener) {
    if (!this[_listeners].has(event)) {
      this[_listeners].set(event, []);
    }
    const list = this[_listeners].get(event);
    if (list.length >= this[_maxListeners]) {
      console.warn(`MaxListenersExceededWarning: ${event}`);
    }
    list.push(listener);
    return this;
  }

  emit(event, ...args) {
    const list = this[_listeners].get(event);
    if (!list) return false;
    list.forEach(fn => fn(...args));
    return true;
  }

  off(event, listener) {
    const list = this[_listeners].get(event);
    if (!list) return this;
    const idx = list.indexOf(listener);
    if (idx !== -1) list.splice(idx, 1);
    return this;
  }
}

const emitter = new EventEmitter();
emitter.on('data', (x) => console.log('received:', x));
emitter.emit('data', 42); // received: 42

// 外部无法直接访问内部状态
console.log(Object.keys(emitter));         // [] ← Symbol 键不可见
console.log(JSON.stringify(emitter));      // {} ← Symbol 键不序列化
// emitter[_listeners]  ← 只有持有 _listeners Symbol 的代码才能访问
```
